```ts
import { PREVIEW_SLIDE_ATTRIBUTE } from './slide-doc';

interface RemovedDisplay {
  style: CSSStyleDeclaration;
  value: string;
  priority: string;
}

function viewOf(document: Document): Window | null {
  return document.defaultView;
}

function computedStyle(
  document: Document,
  element: Element
): CSSStyleDeclaration | null {
  return viewOf(document)?.getComputedStyle(element) ?? null;
}

function computedDisplay(document: Document, element: Element): string {
  return computedStyle(document, element)?.display ?? '';
}

function removeInlineHiding(element: Element): void {
  const style = (element as HTMLElement).style;
  if (!style) return;
  if (style.display === 'none') style.removeProperty('display');
  if (style.visibility === 'hidden' || style.visibility === 'collapse') {
    style.removeProperty('visibility');
  }
  if (style.opacity === '0') style.removeProperty('opacity');
  if (style.getPropertyValue('content-visibility') === 'hidden') {
    style.removeProperty('content-visibility');
  }
}

function removeMatchingDisplayNone(
  document: Document,
  element: Element
): RemovedDisplay[] {
  const removed: RemovedDisplay[] = [];

  const visit = (rules: CSSRuleList): void => {
    for (const rule of Array.from(rules)) {
      if ('selectorText' in rule && 'style' in rule) {
        const css = rule as CSSStyleRule;
        try {
          if (
            element.matches(css.selectorText) &&
            css.style.getPropertyValue('display') === 'none'
          ) {
            removed.push({
              style: css.style,
              value: 'none',
              priority: css.style.getPropertyPriority('display')
            });
            css.style.removeProperty('display');
          }
        } catch {
          // Invalid or unsupported selector; leave it alone.
        }
      }
      if ('cssRules' in rule) {
        try {
          visit((rule as CSSGroupingRule).cssRules);
        } catch {
          // Cross-origin sheet; its rules are opaque.
        }
      }
    }
  };

  for (const sheet of Array.from(document.styleSheets)) {
    try {
      visit(sheet.cssRules);
    } catch {
      // Cross-origin sheet.
    }
  }
  return removed;
}

function forceDisplay(document: Document, element: Element): void {
  if (computedDisplay(document, element) !== 'none') return;
  const removed = removeMatchingDisplayNone(document, element);
  const display = computedDisplay(document, element);
  for (const entry of removed) {
    entry.style.setProperty('display', entry.value, entry.priority);
  }
  (element as HTMLElement).style.setProperty(
    'display',
    display === 'none' ? 'block' : display,
    'important'
  );
}

function forceVisible(document: Document, element: Element): void {
  const style = computedStyle(document, element);
  if (!style) return;
  const target = (element as HTMLElement).style;
  if (style.visibility !== 'visible') {
    target.setProperty('visibility', 'visible', 'important');
  }
  if (style.opacity === '0') {
    target.setProperty('opacity', '1', 'important');
  }
  if (style.getPropertyValue('content-visibility') === 'hidden') {
    target.setProperty('content-visibility', 'visible', 'important');
  }
}

/**
 * Mirrors the renderer reveal pass inside a script-free preview iframe. Only
 * the cloned slide root and its ancestors are touched; authored hiding inside
 * the slide is preserved.
 */
export function revealPreviewSlide(frameDocument: Document): Element | null {
  if (!frameDocument.body) return null;
  const root =
    frameDocument.querySelector(`[${PREVIEW_SLIDE_ATTRIBUTE}]`) ??
    frameDocument.body;
  const chain: Element[] = [];
  for (let current: Element | null = root; current; current = current.parentElement) {
    chain.push(current);
  }

  const wasHidden =
    root.hasAttribute('hidden') || computedDisplay(frameDocument, root) === 'none';
  for (const element of chain) {
    element.removeAttribute('hidden');
    removeInlineHiding(element);
  }
  if (wasHidden) root.classList.add('active');
  for (const element of chain) {
    forceDisplay(frameDocument, element);
    forceVisible(frameDocument, element);
  }
  return root;
}
```

```ts
export interface ViewportSize {
  width: number;
  height: number;
}

export const PLAY_FIT_STORAGE_KEY = 'htmlppt.play-fit';

export function scaleFor(
  box: SlideBox,
  viewport: ViewportSize,
  mode: PlayFitMode
): number {
  if (
    !Number.isFinite(box.width) ||
    !Number.isFinite(box.height) ||
    box.width <= 0 ||
    box.height <= 0 ||
    !Number.isFinite(viewport.width) ||
    !Number.isFinite(viewport.height) ||
    viewport.width <= 0 ||
    viewport.height <= 0
  ) {
    return 1;
  }
  const fit = Math.min(viewport.width / box.width, viewport.height / box.height);
  const fill = Math.max(viewport.width / box.width, viewport.height / box.height);
  const scale = mode === 'fill' ? fill : fit;
  return Number.isFinite(scale) && scale > 0 ? scale : 1;
}

export function parseFitMode(value: string | null | undefined): PlayFitMode {
  return value === 'fill' ? 'fill' : 'fit';
}

export function readFitMode(
  storage?: Pick<Storage, 'getItem'> | null
): PlayFitMode {
  if (!storage) return 'fit';
  try {
    return parseFitMode(storage.getItem(PLAY_FIT_STORAGE_KEY));
  } catch {
    return 'fit';
  }
}

export function writeFitMode(
  mode: PlayFitMode,
  storage?: Pick<Storage, 'setItem'> | null
): void {
  if (!storage) return;
  try {
    storage.setItem(PLAY_FIT_STORAGE_KEY, mode);
  } catch {
    // Blocked storage leaves the session default in place.
  }
}

export function sameBox(a: SlideBox | null, b: SlideBox | null): boolean {
  if (!a || !b) return a === b;
  return (
    Math.abs(a.left - b.left) < 0.5 &&
    Math.abs(a.top - b.top) < 0.5 &&
    Math.abs(a.width - b.width) < 0.5 &&
    Math.abs(a.height - b.height) < 0.5
  );
}

export function measureElementBox(
  element: Element | null,
  viewport: ViewportSize
): SlideBox | null {
  if (!element) return null;
  const rect = element.getBoundingClientRect();
  if (
    !Number.isFinite(rect.left) ||
    !Number.isFinite(rect.top) ||
    !Number.isFinite(rect.width) ||
    !Number.isFinite(rect.height) ||
    rect.width <= 0 ||
    rect.height <= 0
  ) {
    return null;
  }
  if (
    rect.left < -1 ||
    rect.top < -1 ||
    rect.right > viewport.width + 1 ||
    rect.bottom > viewport.height + 1
  ) {
    return null;
  }
  return {
    left: rect.left,
    top: rect.top,
    width: rect.width,
    height: rect.height
  };
}
```

```ts
import { isGroupElement, isHiddenElement, isLockedElement } from './commands/objects';
import { elementAtPath } from './selection';
import {
  TEXT_EDITABLE_SELECTOR,
  textEditBlockFor,
  topLevelBlockFor,
  type EditHintReason
} from './text-edit-block';

export type TextTargetResult =
  | { status: 'editable'; leaf: HTMLElement; empty: boolean }
  | { status: 'blocked'; reason: EditHintReason };

export function isLeafTextElement(element: HTMLElement): boolean {
  if (element.tagName !== 'DIV') return true;
  return Array.from(element.children).every((child) => child.tagName === 'BR');
}

function guardReasonWithin(element: Element, top: Element): EditHintReason | null {
  for (
    let current: Element | null = element;
    current;
    current = current.parentElement
  ) {
    if (isLockedElement(current)) return 'locked';
    if (isHiddenElement(current)) return 'hidden';
    if (isGroupElement(current)) return 'group';
    if (current === top) break;
  }
  return null;
}

function candidatesOf(root: Element): HTMLElement[] {
  const candidates: HTMLElement[] = [];
  if (root.matches(TEXT_EDITABLE_SELECTOR)) candidates.push(root as HTMLElement);
  candidates.push(
    ...Array.from(root.querySelectorAll<HTMLElement>(TEXT_EDITABLE_SELECTOR))
  );
  return candidates;
}

/**
 * Resolves the one leaf that Enter, the Text tab, and the selection toolbar
 * format. A wrapper object resolves its first editable descendant; guarded
 * and empty semantic candidates are skipped, while an empty DIV/TD/TH stays
 * editable so a text box can accept its first text.
 */
export function resolveSelectedTextTarget(
  slide: Element,
  path: readonly number[]
): TextTargetResult {
  const selected = elementAtPath(slide, path);
  const top = elementAtPath(slide, [path[0] ?? -1]);
  if (!selected || !top) return { status: 'blocked', reason: 'not-text' };

  const topGuard = topLevelBlockFor(top);
  if (topGuard) return { status: 'blocked', reason: topGuard };

  let lastGuard: EditHintReason | null = null;
  let sawEmpty = false;
  for (const candidate of candidatesOf(selected)) {
    if (!isLeafTextElement(candidate)) continue;
    const guard = guardReasonWithin(candidate, top);
    if (guard) {
      lastGuard = guard;
      continue;
    }
    const empty = !candidate.textContent?.trim();
    if (
      empty &&
      candidate.tagName !== 'DIV' &&
      candidate.tagName !== 'TD' &&
      candidate.tagName !== 'TH'
    ) {
      sawEmpty = true;
      continue;
    }
    return { status: 'editable', leaf: candidate, empty };
  }

  if (lastGuard) return { status: 'blocked', reason: lastGuard };
  if (sawEmpty) return { status: 'blocked', reason: 'empty' };
  return {
    status: 'blocked',
    reason: textEditBlockFor(selected) ?? 'not-text'
  };
}
```

```ts
export type TextShortcutAction = {
  type: 'bold' | 'italic' | 'underline';

/**
 * Text commands are separate from mapShortcut: contenteditable targets own
 * normal shortcuts, but B/I/U must still reach the text command layer. The
 * caller decides whether a text target exists and whether a form field owns
 * the key.
 */
export function mapTextShortcut(
  event: ShortcutEventLike
): TextShortcutAction | null {
  if (event.isComposing || event.repeat) return null;
  if (event.altKey || event.shiftKey) return null;
  if (!primaryModifier(event)) return null;
  const key = normalizedKey(event);
  if (key === 'b') return { type: 'bold' };
  if (key === 'i') return { type: 'italic' };
  if (key === 'u') return { type: 'underline' };
  return null;
}
```
